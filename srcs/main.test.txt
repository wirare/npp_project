#include <cstdint>
#include <cuda_runtime.h>
#include <nppcore.h>
#include <nppi.h>
#include <nppi_support_functions.h>
#include <string>
#include "stb_image.h"
#include "stb_image_write.h"

Npp8u *load_image(const std::string name, int *width, int *height, int *stride)
{
    int channels_in_file = 0;
    uint8_t *stbi_data = stbi_load(name.c_str(), width, height, &channels_in_file, 3);
    if (!stbi_data)
        return nullptr;

    Npp8u *nppi_data = nppiMalloc_8u_C3(*width, *height, stride);
    if (!nppi_data)
	{
        stbi_image_free(stbi_data);
        return nullptr;
    }

    const size_t rowBytes = static_cast<size_t>(*width) * 3;

    cudaError_t err = cudaMemcpy2D(
        nppi_data,
        static_cast<size_t>(*stride),
        stbi_data,
        rowBytes,                    
        rowBytes,                    
        static_cast<size_t>(*height),
        cudaMemcpyHostToDevice
    );

    stbi_image_free(stbi_data);

    if (err != cudaSuccess)
	{
        nppiFree(nppi_data);
        return nullptr;
    }

    return nppi_data;
}

void write_image(const std::string name, Npp8u *nppi_data, int width, int height, int stride)
{
	uint8_t *stbi_data = (uint8_t *)malloc(width * height * 3 * sizeof(uint8_t));

	const size_t rowBytes = static_cast<size_t>(width) * 3;

	cudaError_t err = cudaMemcpy2D(
		stbi_data,
		rowBytes,
		nppi_data,
		static_cast<size_t>(stride),
		rowBytes,
		static_cast<size_t>(height),
		cudaMemcpyDeviceToHost
	);

	if (err != cudaSuccess)
	{
		free(stbi_data);
		return;
	}

	stbi_write_png(name.c_str(), width, height, 3, stbi_data, rowBytes);
	free(stbi_data);
}

int main(int ac, char **av)
{
	int deviceId = 0;
	cudaSetDevice(deviceId);

	cudaStream_t stream = 0;
	nppSetStream(stream);

	NppStreamContext nppStreamCtx;
	nppGetStreamContext(&nppStreamCtx);

	int width, height, stride_A, stride_B;

	Npp8u *buff_A = load_image(av[1], &width, &height, &stride_A);
	if (buff_A == nullptr)
		return 1;

	Npp8u *buff_B = nppiMalloc_8u_C3(width, height, &stride_B);
	if (!buff_B) 
	{
        nppiFree(buff_A);
        return 1;
    }

	NppiSize full_ROI = {width, height};

	NppStatus err = nppiFilterSobelVert_8u_C3R_Ctx(buff_A, stride_A, buff_B, stride_B, full_ROI, nppStreamCtx);

	if (err != NPP_SUCCESS)
	{
		nppiFree(buff_A);
		nppiFree(buff_B);
		return 1;
	}

	write_image(av[2], buff_B, width, height, stride_B);
	nppiFree(buff_A);
	nppiFree(buff_B);
	return 0;
}
